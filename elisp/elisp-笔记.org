  ;; 在 elisp 中，字符是整数，整数是不是字符取决于用法

* 列表
#+begin_src emacs-lisp
  (consp '(a . b))            ;; 检测是否是 cons 单元
  (atom 'a)                   ;; 检测是否是原子
  (listp '(a b c))            ;; 检测是否是列表
  (null nil)                  ;; 检测是否为 nil
  (proper-list-p '(a b c))    ;; 检测如果是列表则返回列表的长度，不是则返回 nil

  (car '(a b c))             ;; 返回第一个元素 a，car-safe 是 car 的安全版本
  (cdr '(a b c))             ;; 返回剩下的 (b c)，cdr-safe 是 cdr 的安全版本
  (pop arr)                  ;; 删除 arr 中的第一个元素
  (nth 2 '(a b c))           ;; 返回列表下标的第 2 个元素
  (nthcdr 2 '(a b c d e))    ;; 返回列表下标第二个以后的元素 (c d e)
  (last '(a b c d) 2)        ;; 返回最后两个 (c d)
  (safe-length '(a b c))     ;; 返回列表的长度
  (butlast '(a b c d) 2)     ;; 删除最后两个元素，剩下 (a b)

  (cons 'a 'b)                ;; 构建一个简单的 (a . b)
  (list 1 2 '(3 4))           ;; 构建列表 (1 2 (3 4))
  (make-list 3 'x)            ;; 构建列表 (x x x)
  (append '(a b) '(c d))      ;; 构建新列表 (a b c d)
  (number-sequence 3 10 2)    ;; 创建一个 3 到 10 的列表，间隔为 2，返回 (3 5 7 9)

  (setq arr '(b c d))
  (push 'a arr)            ;; 将 a 插入到列表开头，返回 (a b c)
  (add-to-list 'arr 'a)    ;; 将 a 插入到列表开头，返回 (a b c)，如果元素已存在，则不会增加元素

  (setcar arr 'a)         ;; 修改列表的第一个元素为 a
  (setcdr arr '(f g))     ;; 修改列表的第一个元素以后的元素为 (f g)
  (nconc arr '(x y z))    ;; 后续接上 (x y z) 返回 (b c d x y z)，重复执行会在结尾不停的续接

  (memq 'c '(a b c d))     ;; 只保留元素 c 后面的，返回 (c d)
  (delq 'c '(a b c d))     ;; 删除元素 c
  (remq 'a '(a b a c a))   ;; 删除列表中所有的元素 a，返回 (b c)

  ;; 关联列表 alist 记录了从键到值的映射，它是有 cons 组成的列表
  (setq my-alist '((name . "Alice")    ;; 定义一个简单的 alist
                 (age . 25)
                 (city . "Beijing")))
  (assoc 'city my-alist)         ;; 根据 key 查找
  (rassoc "Alice" my-alist)      ;; 根据值来查找

  ;; plist 是成对元素的列表
  (setq my-plist '(pine cones num (1 2 3) color "blue"))
  (plist-get my-plist 'num)             ;; 获取 num 的值
  (plist-put my-plist 'num '(5 6 7))    ;; 把 num 的值替换成 (5 6 7)
#+end_src

#+begin_src emacs-lisp
  (let ((x 1)    ;; 局部绑定，x y 作用域只在 let 的括号范围内生效

** 条件语句
#+begin_src emacs-lisp
  (if condition      ;; 当条件满足时可执行语句 statement1，否则执行 statement2
      (statement1)
    (statement2))

  (when condition    ;; 当条件满足时可同时执行多条语句，返回最后一条语句
    (statement1)
    (statement2)
    (statement3)
    ...)

  (unless condition    ;; 当条件不满足时可同时执行多条语句，返回最后一条语句
    (statement1)
    (statement2)
    (statement3)
    ...)

  (cond                 ;; 条件满足哪个就执行哪个，都没有满足的就执行最后语句默认语句
   (condition1 statements...)
   (condition2 statements...)
   ...
   (t default-statements...))

  ;; 简单的例子：判断奇偶数 --------------------------------------------------

  (if (= 0 (% 4 2))
      (message "是偶数")
    (message "是奇数"))

  (when (= 0 (% 4 2))
    (message "是偶数"))
  (when (/= 0 (% 4 2))
    (message "是奇数"))

  (unless (/= 0 (% 4 2))
    (message "是偶数"))
  (unless (= 0 (% 5 2))
    (message "是奇数"))

  (cond
   ((/= 0 (% 4 2)) (message "是奇数"))
   ((= 0 (% 4 2)) (message "是偶数"))
    (t (message "不是一个数")))
#+end_src

** pcase 宏
#+begin_src emacs-lisp
  (pcase 要匹配的值
    (模式1 执行代码1...)
    (模式2 执行代码2...)
    ...)

  (pcase (% 5 2)
    (0 "偶数")    ;; 结果是 0 返回 "偶数"
    (1 "奇数")    ;; 结果是 1 返回 "奇数"
    (_ "其他"))   ;; 结果是其他返回 "其他"，_ 符号表示其余结果

  (pcase (% 5 2)                ;; pred 表示用 (% 5 2) 的值 1 去匹配下面的语句
    ((pred (= 0)) "是偶数")     ;; 等价于 (= 0 1) 注意顺序，0 在前面
    ((pred (= 1)) "是奇数"))    ;; 等价于 (= 1 1)

  (pcase (* 4 2)                          ;; 使用 lambda 函数在逻辑上会更好的理解
    ((pred (lambda (x) (> x 0))) "是正数")
    ((pred (lambda (x) (< x 0))) "是负数"))

  (pcase '(0 1)    ;; 利用反引号的特性进行匹配
    ('(0 0) '在原点)
    (`(0 ,y) '在y轴上)
    (`(,x 0) '在x轴上)
    (_ '在坐标轴内))
#+end_src
** 迭代
#+begin_src emacs-lisp
  (dotimes (i 5)    ;; i 从 0 开始，每循环 1 次加 1，最后返回 nil
    (message "第 %d 次循环" i))

  (dolist (i (number-sequence 0 4 1))    ;; 这里 number-sequence 表示 0 到 4 的范围，步长是 1
    (message "第 %d 次循环" i))
  (dolist (i '(0 1 2 3 4))
     (message "第 %d 次循环" i))

  (setq i 0)
  (while (< i 5)    ;; while 最后永远返回的是 nil
    (message "第 %d 次循环" i)
    (setq i (+ i 1)))
#+end_src
