  #b101100    ;; #b 表示 2 进制

  (round -1.3)       ;; 四舍五入，fround 函数返回的是浮点数

* 字符串和字符
#+begin_src emacs-lisp
  ;; 在 elisp 中，字符是整数，整数是不是字符取决于用法
  ?a    ;; 表示字符 a，如果不加 “?”,则表示的是变量 a
  (stringp "he")            ;; 判断是不是字符串
  (string-or-null-p nil)    ;; 如果是字符串或 nil，则返回 t
  (char-or-string-p 12)     ;; 是字符串或字符（即整数）则返回 t

  (make-string 5 ?a)          ;; 创建 5 个字符 a 组成的字符串 "aaaaa"
  (string ?h ?e ?l)           ;; 组成字符串 "hel"
  (aref "hello" 1)            ;; 取出字符下标为 1 的字符串，从 0 开始

  (substring "hello" 1 4)          ;; 从下标 0 开始，截取第 1 到 3 个字符组成字符串 "ell"
  (substring "hello" -3 -1)        ;; 倒数第 3 到倒数第 2 个 组成字符串 “ll”
  (substring [a b c d e f] 1 4)    ;; 支持向量，返回 [b c d]

  (concat "he" "-ad" "c")       ;; 组成字符串 “he-adc”
  (concat "he" (list 97 98))    ;; 可以列表组成字符串 "heab"

  (split-string "hello15280740096ad" "[^0-9]+")    ;; 用正则表达式匹配手机号

  (aset "hello" 1 ?x)      ;; 替换下标为 1 的字符为 x
  (aset [a b c d] 1 'x)    ;; 替换下标为 1 的字符为 x
  (store-substring "hello world" 6 "Emacs")     ;; 替换下标从 6 开始变成 "hello Emacs"

  (char-equal ?x ?x)     ;; 字符比较，相同的字符返回 t
  (string= "ab" "ab")    ;; 字符串比较，相同的字符串返回 t
  (string< "ab" "ba")    ;; 比较第一对不匹配的字符

  (number-to-string 123)      ;; 数字转化为字符串
  (string-to-number "123")    ;; 字符串转数字
  (string-to-char "abc")      ;; 返回字符串中的第一个字符

  (downcase "Hello World")    ;; 把大写转化为小写
  (upcase "hello world")      ;; 把小写转化为大写

  (format "hello %s" "world")            ;; 类似于 c 语言的 printf 格式的用法，返回值为 "hello world"
  (format-message "hello %s" "world")    ;; 用法与上面的基本一致，更安全，优秀推荐使用
#+end_src

* 列表
#+begin_src emacs-lisp
  (consp '(a . b))            ;; 检测是否是 cons 单元
  (atom 'a)                   ;; 检测是否是原子
  (listp '(a b c))            ;; 检测是否是列表
  (null nil)                  ;; 检测是否为 nil
  (proper-list-p '(a b c))    ;; 检测如果是列表则返回列表的长度，不是则返回 nil

  (car '(a b c))             ;; 返回第一个元素 a，car-safe 是 car 的安全版本
  (cdr '(a b c))             ;; 返回剩下的 (b c)，cdr-safe 是 cdr 的安全版本
  (pop arr)                  ;; 删除 arr 中的第一个元素
  (nth 2 '(a b c))           ;; 返回列表下标的第 2 个元素
  (nthcdr 2 '(a b c d e))    ;; 返回列表下标第二个以后的元素 (c d e)
  (last '(a b c d) 2)        ;; 返回最后两个 (c d)
  (safe-length '(a b c))     ;; 返回列表的长度
  (butlast '(a b c d) 2)     ;; 删除最后两个元素，剩下 (a b)

  (cons 'a 'b)                ;; 构建一个简单的 (a . b)
  (list 1 2 '(3 4))           ;; 构建列表 (1 2 (3 4))
  (make-list 3 'x)            ;; 构建列表 (x x x)
  (append '(a b) '(c d))      ;; 构建新列表 (a b c d)
  (number-sequence 3 10 2)    ;; 创建一个 3 到 10 的列表，间隔为 2，返回 (3 5 7 9)

  (setq arr '(b c d))
  (push 'a arr)            ;; 将 a 插入到列表开头，返回 (a b c)
  (add-to-list 'arr 'a)    ;; 将 a 插入到列表开头，返回 (a b c)，如果元素已存在，则不会增加元素

  (setcar arr 'a)         ;; 修改列表的第一个元素为 a
  (setcdr arr '(f g))     ;; 修改列表的第一个元素以后的元素为 (f g)
  (nconc arr '(x y z))    ;; 后续接上 (x y z) 返回 (b c d x y z)，重复执行会在结尾不停的续接

  (memq 'c '(a b c d))    ;; 只保留元素 c 后面的，返回 (c d)
  (delq 'c '(a b c d))    ;; 删除元素 c
#+end_src
