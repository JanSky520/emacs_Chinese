* 常用函数
** 缓冲区
#+begin_src emacs-lisp
  (buffer-name)            ; 显示当前 buffer 的名字
  (buffer-file-name)       ; 显示当前 buffer 完整路径
  (current-buffer)         ; 显示当前 buffer
  (other-buffer)           ; 显示前一个 buffer 的名字
  (buffer-size)            ; 显示当前缓冲区的字符数
  (point)                  ; 表示从缓冲区开头到光标所在位置的字符数量
  (point-min)              ; 返回当前缓冲区中 point 的最小允许值
  (point-max)              ; 返回当前缓冲区的最大可访问位置 + 1
  (beginning-of-buffer)    ; 将光标移动到文件开头
  (mark-whole-buffer)      ; 选中当前文件的全部内容
#+end_src
** interactive 函数
#+begin_src emacs-lisp
  (interactive)        ; 在设置的 defun 中添加这段函数可以使函数变成交互式的
  (interactive "p")    ; 在启用函数前，按 C-u 输入一个参数，在 M-x 可以把参数传入函数
  (interactive "n")    ; 接收一个数字参数
  (interactive "s")    ; 接收一个字符串参数
  (interactive "b")    ; 接收一个 buffer 参数
  (interactive "B")    ; 缓冲区可以不存在
#+end_src

* 常用组合
** 缓冲区
#+begin_src emacs-lisp
  (buffer-name (current-buffer))     ; 显示当前 buffer 的名字
  (switch-to-buffer (other-buffer))  ; 跳转到上一个 buffer
  (set-buffer "*scratch*")           ; 后台切换到 *scratch* 缓冲区， switch 相对显示，set 看不见
#+end_src

* 函数实战
** 缓冲区
#+begin_src emacs-lisp
  (defun buffer-test()
    "简单的输出当前文件的一些内容"
    (interactive)
    (message "当前缓冲区：%s" (buffer-name) (point))
    (message "当前文件名：%s" (buffer-file-name))
    (message "当前文件字符数：%d，当前光标所在位置：%d" (point-max) (point)))

  (save-excursion                       ; 保存当前状态，执行完代码后恢复原始状态
    (goto-char (point-max))             ; 跳转到文件末尾
    (insert "hello"))                   ; 插入字符串后自动返回原处
#+end_src
** interactive 函数
#+begin_src emacs-lisp
  (defun interactive-test(x y)
    (interactive "n输入x: \nn输入y：")    ; \n 是换行的意思
    (message "x * y = %d" (* x y)))

  (defun beginning-of-buffer-test()
    (interactive)
    (push-mark)                     ; 记录当前光标位置
    (goto-char (point-min)))        ; 跳转到文件开头

  (defun mark-whole-buffer-test()
    (interactive)
    (push-mark (point-max))         ; 记录光标的最大值
    (goto-char (point-min))         ; 跳转到文件开头
    (exchange-point-and-mark))      ; 交换当前点 (point) 和标记 (mark) 的位置

  (defun seek-buffer(tmp)
    (interactive "s输入寻找的缓冲区：")
    (if (get-buffer tmp)
        (message "缓冲区存在")
      (message "缓冲区不存在")))
#+end_src
